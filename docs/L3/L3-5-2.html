2.7.2 1.2 샘플 데이터 생성 (데모용)

실제 MIMIC-CXR 데이터 접근이 불가능한 경우를 위해, 구조적으로 유사한 샘플 데이터를 생성합니다.

Code 
# 샘플 데이터 생성 (데모용 - 실제 분석 시 실제 데이터 사용)
def create_sample_dataset(n_samples: int = 1000) -> pd.DataFrame:
    """
    MIMIC-CXR 구조를 모방한 샘플 데이터셋 생성
    실제 분석에서는 이 함수 대신 실제 데이터를 로드합니다.
    """
    np.random.seed(42)
    
    # 샘플 판독문 템플릿
    report_templates = [
        "The lungs are clear. No focal consolidation, pleural effusion or pneumothorax. "
        "The cardiomediastinal silhouette is normal. No acute osseous abnormalities.",
        
        "There is a small left pleural effusion. Patchy opacity in the right lower lobe "
        "may represent atelectasis or infection. Heart size is mildly enlarged.",
        
        "Bilateral interstitial opacities consistent with pulmonary edema. "
        "Cardiomegaly noted. Small bilateral pleural effusions present.",
        
        "Large right-sided pneumothorax with near-complete collapse of the right lung. "
        "Mediastinal shift to the left. Recommend urgent intervention.",
        
        "Multifocal pneumonia with consolidation in bilateral lower lobes. "
        "Moderate bilateral pleural effusions. Recommend clinical correlation.",
        
        "No acute cardiopulmonary abnormality. Stable appearance compared to prior.",
        
        "Mild cardiomegaly. Clear lung fields. No pleural effusion or pneumothorax.",
        
        "Diffuse bilateral airspace opacities concerning for ARDS. "
        "Endotracheal tube in appropriate position. Critical findings."
    ]
    
    # 중증도 라벨 (Critical: 1, Non-critical: 0)
    severity_mapping = [0, 0, 1, 1, 1, 0, 0, 1]
    
    data = []
    for i in range(n_samples):
        template_idx = np.random.randint(0, len(report_templates))
        report = report_templates[template_idx]
        if np.random.random() > 0.5:
            report = report.replace("right", "left") if "right" in report else report
        
        severity = severity_mapping[template_idx]
        if np.random.random() < 0.1:
            severity = 1 - severity
        
        data.append({
            'subject_id': f'p{10000000 + i}',
            'study_id': f's{50000000 + i}',
            'dicom_id': f'd{np.random.randint(10000000, 99999999)}',
            'ViewPosition': np.random.choice(['PA', 'AP', 'LATERAL'], p=[0.5, 0.4, 0.1]),
            'report': report,
            'severity': severity,
            'split': np.random.choice(['train', 'validate', 'test'], p=[0.7, 0.15, 0.15])
        })
    
    return pd.DataFrame(data)

# 샘플 데이터 생성
df_sample = create_sample_dataset(n_samples=2000)

🔍 학습 포인트 #1: 멀티모달 데이터 탐색의 중요성

멀티모달 의료 AI에서 데이터 탐색은 각 모달리티(영상, 텍스트)의 특성을 개별적으로 이해하고, 두 모달리티 간의 상관관계를 파악하는 것이 핵심입니다.

  • 영상 데이터: 촬영 방향(PA/AP)에 따라 심장 크기가 다르게 보일 수 있어 전처리 시 고려 필요
  • 텍스트 데이터: 판독문의 길이와 사용된 의학 용어가 중증도와 상관관계를 가짐

class ImagePreprocessor:
    """
    흉부 X-ray 영상 전처리 클래스
    """
    def __init__(self, image_size: int = 224, augment: bool = False):
        self.image_size = image_size
        self.augment = augment
        
        self.base_transform = transforms.Compose([
            transforms.Resize((image_size, image_size)),
            transforms.ToTensor(),
            transforms.Normalize(
                mean=[0.485, 0.456, 0.406],
                std=[0.229, 0.224, 0.225]
            )
        ])
        
        self.train_transform = transforms.Compose([
            transforms.Resize((image_size + 32, image_size + 32)),
            transforms.RandomCrop(image_size),
            transforms.RandomHorizontalFlip(p=0.5),
            transforms.RandomRotation(degrees=10),
            transforms.ColorJitter(brightness=0.2, contrast=0.2),
            transforms.ToTensor(),
            transforms.Normalize(
                mean=[0.485, 0.456, 0.406],
                std=[0.229, 0.224, 0.225]
            )
        ])
    
    def __call__(self, image: Image.Image) -> torch.Tensor:
        if self.augment:
            return self.train_transform(image)
        return self.base_transform(image)

def create_synthetic_xray(size: int = 256) -> Image.Image:
    np.random.seed(42)
    base = np.random.normal(128, 30, (size, size)).astype(np.uint8)
    y, x = np.ogrid[:size, :size]
    lung_left = ((x - size*0.35)**2 + (y - size*0.5)**2) < (size*0.25)**2
    lung_right = ((x - size*0.65)**2 + (y - size*0.5)**2) < (size*0.25)**2
    base[lung_left] = np.clip(base[lung_left] - 40, 0, 255)
    base[lung_right] = np.clip(base[lung_right] - 40, 0, 255)
    rgb_image = np.stack([base, base, base], axis=-1)
    return Image.fromarray(rgb_image)

        }

class ImageEncoder(nn.Module):
    """영상 인코더: ResNet-50 백본"""
    def __init__(self, pretrained: bool = True, freeze: bool = True):
        super().__init__()
        self.backbone = models.resnet50(pretrained=pretrained)
        self.feature_dim = self.backbone.fc.in_features
        self.backbone.fc = nn.Identity()
        
        if freeze:
            for param in self.backbone.parameters():
                param.requires_grad = False
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        features = self.backbone(x)
        return features


class TextEncoder(nn.Module):
    """텍스트 인코더: BioBERT 백본"""
    def __init__(self, model_name: str = 'dmis-lab/biobert-base-cased-v1.2', freeze: bool = True):
        super().__init__()
        self.bert = AutoModel.from_pretrained(model_name)
        self.feature_dim = self.bert.config.hidden_size
        
        if freeze:
            for param in self.bert.parameters():
                param.requires_grad = False
    
    def forward(self, input_ids: torch.Tensor, attention_mask: torch.Tensor) -> torch.Tensor:
        outputs = self.bert(input_ids=input_ids, attention_mask=attention_mask)
        cls_output = outputs.last_hidden_state[:, 0, :]
        return cls_output


class BaselineConcatModel(nn.Module):
    """Baseline: 단순 Concatenation 융합 모델"""
    def __init__(self, num_classes: int = 2, freeze_encoders: bool = True):
        super().__init__()
        
        self.image_encoder = ImageEncoder(pretrained=True, freeze=freeze_encoders)
        self.text_encoder = TextEncoder(freeze=freeze_encoders)
        
        combined_dim = self.image_encoder.feature_dim + self.text_encoder.feature_dim
        
        self.classifier = nn.Sequential(
            nn.Linear(combined_dim, 512),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(512, num_classes)
        )
    
    def forward(
        self,
        image: torch.Tensor,
        input_ids: torch.Tensor,
        attention_mask: torch.Tensor
    ) -> torch.Tensor:
        image_features = self.image_encoder(image)
        text_features = self.text_encoder(input_ids, attention_mask)
        combined = torch.cat([image_features, text_features], dim=1)
        logits = self.classifier(combined)
        return logits

# 학습 유틸리티 함수
def train_epoch(
    model: nn.Module,
    dataloader: DataLoader,
    criterion: nn.Module,
    optimizer: torch.optim.Optimizer,
    device: torch.device
) -> Tuple[float, float]:
    model.train()
    total_loss = 0.0
    correct = 0
    total = 0
    
    pbar = tqdm(dataloader, desc="Training")
    for batch in pbar:
        images = batch['image'].to(device)
        input_ids = batch['input_ids'].to(device)
        attention_mask = batch['attention_mask'].to(device)
        labels = batch['label'].to(device)
        
        optimizer.zero_grad()
        outputs = model(images, input_ids, attention_mask)
        loss = criterion(outputs, labels)
        
        loss.backward()
        optimizer.step()
        
        total_loss += loss.item()
        _, predicted = outputs.max(1)
        total += labels.size(0)
        correct += predicted.eq(labels).sum().item()
        
        pbar.set_postfix({'loss': f"{loss.item():.4f}"})
    
    avg_loss = total_loss / len(dataloader)
    accuracy = 100. * correct / total
    
    return avg_loss, accuracy


def evaluate(
    model: nn.Module,
    dataloader: DataLoader,
    criterion: nn.Module,
    device: torch.device
) -> Tuple[float, float, np.ndarray, np.ndarray]:
    model.eval()
    total_loss = 0.0
    all_preds = []
    all_labels = []
    all_probs = []
    
    with torch.no_grad():
        for batch in tqdm(dataloader, desc="Evaluating"):
            images = batch['image'].to(device)
            input_ids = batch['input_ids'].to(device)
            attention_mask = batch['attention_mask'].to(device)
            labels = batch['label'].to(device)
            
            outputs = model(images, input_ids, attention_mask)
            loss = criterion(outputs, labels)
            
            total_loss += loss.item()
            probs = F.softmax(outputs, dim=1)[:, 1]
            _, predicted = outputs.max(1)
            
            all_preds.extend(predicted.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())
            all_probs.extend(probs.cpu().numpy())
    
    avg_loss = total_loss / len(dataloader)
    accuracy = 100. * np.mean(np.array(all_preds) == np.array(all_labels))
    
    return avg_loss, accuracy, np.array(all_probs), np.array(all_labels)

class CrossAttention(nn.Module):
    """
    Cross-Attention 모듈
    Query: 한 모달리티의 특징
    Key, Value: 다른 모달리티의 특징
    """
    def __init__(
        self,
        query_dim: int,
        key_dim: int,
        embed_dim: int = 256,
        num_heads: int = 8,
        dropout: float = 0.1
    ):
        super().__init__()
        
        self.embed_dim = embed_dim
        self.num_heads = num_heads
        self.head_dim = embed_dim // num_heads
        
        assert self.head_dim * num_heads == embed_dim, "embed_dim must be divisible by num_heads"
        
        self.query_proj = nn.Linear(query_dim, embed_dim)
        self.key_proj = nn.Linear(key_dim, embed_dim)
        self.value_proj = nn.Linear(key_dim, embed_dim)
        self.output_proj = nn.Linear(embed_dim, embed_dim)
        
        self.layer_norm = nn.LayerNorm(embed_dim)
        self.dropout = nn.Dropout(dropout)
        self.scale = self.head_dim ** -0.5
    
    def forward(
        self,
        query: torch.Tensor,
        key: torch.Tensor,
        value: torch.Tensor = None,
        return_attention: bool = False
    ) -> Tuple[torch.Tensor, Optional[torch.Tensor]]:
        
        if value is None:
            value = key
        
        if query.dim() == 2:
            query = query.unsqueeze(1)
        if key.dim() == 2:
            key = key.unsqueeze(1)
            value = value.unsqueeze(1)
        
        batch_size = query.size(0)
        
        Q = self.query_proj(query)
        K = self.key_proj(key)
        V = self.value_proj(value)
        
        Q = Q.view(batch_size, -1, self.num_heads, self.head_dim).transpose(1, 2)
        K = K.view(batch_size, -1, self.num_heads, self.head_dim).transpose(1, 2)
        V = V.view(batch_size, -1, self.num_heads, self.head_dim).transpose(1, 2)
        
        attention_scores = torch.matmul(Q, K.transpose(-2, -1)) * self.scale
        attention_weights = F.softmax(attention_scores, dim=-1)
        attention_weights = self.dropout(attention_weights)
        
        context = torch.matmul(attention_weights, V)
        context = context.transpose(1, 2).contiguous().view(batch_size, -1, self.embed_dim)
        
        output = self.output_proj(context)
        output = self.layer_norm(output)
        
        output = output.squeeze(1) if output.size(1) == 1 else output
        
        if return_attention:
            return output, attention_weights
        return output, None

class CrossAttentionFusionModel(nn.Module):
    """
    Cross-Attention 기반 멀티모달 융합 모델
    양방향 Cross-Attention:
    1. Image-to-Text: 영상 정보가 텍스트를 참조
    2. Text-to-Image: 텍스트 정보가 영상을 참조
    """
    def __init__(
        self,
        num_classes: int = 2,
        embed_dim: int = 256,
        num_heads: int = 8,
        freeze_encoders: bool = True
    ):
        super().__init__()
        
        self.image_encoder = ImageEncoder(pretrained=True, freeze=freeze_encoders)
        self.text_encoder = TextEncoder(freeze=freeze_encoders)
        
        image_dim = self.image_encoder.feature_dim
        text_dim = self.text_encoder.feature_dim
        
        self.image_proj = nn.Linear(image_dim, embed_dim)
        self.text_proj = nn.Linear(text_dim, embed_dim)
        
        self.cross_attn_i2t = CrossAttention(
            query_dim=embed_dim, key_dim=embed_dim,
            embed_dim=embed_dim, num_heads=num_heads
        )
        self.cross_attn_t2i = CrossAttention(
            query_dim=embed_dim, key_dim=embed_dim,
            embed_dim=embed_dim, num_heads=num_heads
        )
        
        self.fusion = nn.Sequential(
            nn.Linear(embed_dim * 4, embed_dim * 2),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(embed_dim * 2, embed_dim),
            nn.ReLU(),
            nn.Dropout(0.2)
        )
        
        self.classifier = nn.Linear(embed_dim, num_classes)
        self.attention_weights_i2t = None
        self.attention_weights_t2i = None
    
    def forward(
        self,
        image: torch.Tensor,
        input_ids: torch.Tensor,
        attention_mask: torch.Tensor,
        return_attention: bool = False
    ) -> torch.Tensor:
        
        image_features = self.image_encoder(image)
        text_features = self.text_encoder(input_ids, attention_mask)
        
        image_proj = self.image_proj(image_features)
        text_proj = self.text_proj(text_features)
        
        image_attended, attn_i2t = self.cross_attn_i2t(
            query=image_proj, key=text_proj, return_attention=True
        )
        
        text_attended, attn_t2i = self.cross_attn_t2i(
            query=text_proj, key=image_proj, return_attention=True
        )
        
        self.attention_weights_i2t = attn_i2t
        self.attention_weights_t2i = attn_t2i
        
        combined = torch.cat([
            image_proj,
            text_proj,
            image_attended,
            text_attended
        ], dim=1)
        
        fused = self.fusion(combined)
        logits = self.classifier(fused)
        
        if return_attention:
            return logits, (attn_i2t, attn_t2i)
        return logits

🔍 학습 포인트 #5: Cross-Attention의 작동 원리

Cross-Attention은 두 모달리티 간의 상호작용을 명시적으로 학습합니다:

수학적 정의: \[\text{CrossAttn}(Q, K, V) = \text{softmax}\left(\frac{QK^T}{\sqrt{d_k}}\right)V\]

Image-to-Text Attention: - Query: 영상 특징 - Key/Value: 텍스트 특징 - 의미: “영상의 어떤 부분이 텍스트의 어떤 정보와 관련있는가?”

Text-to-Image Attention: - Query: 텍스트 특징 - Key/Value: 영상 특징 - 의미: “’폐렴’이라는 단어가 영상의 어떤 영역에 해당하는가?”

def unfreeze_layers(model, num_layers: int = 2):
    """인코더의 상위 레이어를 점진적으로 해제"""
    resnet_layers = [
        model.image_encoder.backbone.layer4,
        model.image_encoder.backbone.layer3,
        model.image_encoder.backbone.layer2,
        model.image_encoder.backbone.layer1,
    ]
    
    for i, layer in enumerate(resnet_layers[:num_layers]):
        for param in layer.parameters():
            param.requires_grad = True
        print(f"  ✓ ResNet layer{4-i} 해제")
    
    bert_layers = list(model.text_encoder.bert.encoder.layer)
    for layer in bert_layers[-num_layers:]:
        for param in layer.parameters():
            param.requires_grad = True
    print(f"  ✓ BERT 상위 {num_layers} 레이어 해제")
    
    return model
