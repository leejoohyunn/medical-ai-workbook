2 🧬 L3-4 BioBERT 기반 임상 텍스트 분류

2.1 🧭 학습 목표 (Learning Outcomes)

이 워크북을 통해 학습자는 다음을 실습하게 됩니다:

  • 의료 텍스트 전처리 및 도메인 특화 정규화 기법 적용
  • Word2Vec → BERT → BioBERT → BioMistral로 이어지는 점진적 모델 개선 경험
  • Full Fine-tuning, LoRA, QLoRA 등 다양한 파인튜닝 전략 비교 및 구현
  • Attention Weights, Integrated Gradients를 활용한 모델 해석
  • 임상 텍스트 분류 모델의 외부 검증 및 실무 배포 가능성 탐색

2.2 🧩 실습 개요

  • 데이터: MIMIC-CXR Report (Non-image Text subset)
  • 목표: 흉부 X-ray 판독문으로부터 폐 질환(Pneumonia, Atelectasis, Cardiomegaly 등) 다중 레이블 분류
  • 사용 기술: Transformers, BioBERT, PEFT (LoRA/QLoRA), PyTorch, Captum
  • 제약 조건:
    1. GPU 메모리 제한(16GB 이하)을 고려한 효율적 학습 전략 필수
    2. 의료 도메인 특화 사전학습 모델 활용 필수
    3. 모델 해석 가능성(Explainability) 확보 필수

def extract_sections(text):
    """
    판독문에서 FINDINGS와 IMPRESSION 섹션 추출
    IMPRESSION이 더 응축된 진단 정보를 포함
    """
    text_lower = text.lower()

    # IMPRESSION 섹션 추출
    impression_match = re.search(
        r'impression[:/]?\s*(.+?)(?=findings|$)',
        text_lower,
        re.DOTALL
    )

    # FINDINGS 섹션 추출
    findings_match = re.search(
        r'findings[:/]?\s*(.+?)(?=impression|$)',
        text_lower,
        re.DOTALL
    )

    impression = impression_match.group(1).strip() if impression_match else ''
    findings = findings_match.group(1).strip() if findings_match else text_lower

    return {
        'findings': findings,
        'impression': impression,
        'combined': f"{findings} {impression}".strip()
    }

# 의료 도메인에서 중요한 민감도(Sensitivity)와 특이도(Specificity)
def calculate_clinical_metrics(y_true, y_pred, label_names):
    """
    임상적으로 중요한 메트릭 계산
    - Sensitivity (TPR): 실제 양성을 양성으로 예측
    - Specificity (TNR): 실제 음성을 음성으로 예측
    - PPV (Precision): 양성 예측의 정확도
    - NPV: 음성 예측의 정확도
    """
    results = []

    for i, label in enumerate(label_names):
        cm = multilabel_confusion_matrix(y_true, y_pred)[i]
        tn, fp, fn, tp = cm.ravel()

        sensitivity = tp / (tp + fn) if (tp + fn) > 0 else 0
        specificity = tn / (tn + fp) if (tn + fp) > 0 else 0
        ppv = tp / (tp + fp) if (tp + fp) > 0 else 0
        npv = tn / (tn + fn) if (tn + fn) > 0 else 0

        results.append({
            'Label': label,
            'Sensitivity': sensitivity,
            'Specificity': specificity,
            'PPV': ppv,
            'NPV': npv
        })

    return pd.DataFrame(results)

def get_attention_weights(model, tokenizer, text, device):
    """
    BERT 모델의 Attention weights 추출
    """
    model.eval()

    # 토큰화
    inputs = tokenizer(
        text,
        return_tensors='pt',
        truncation=True,
        max_length=256,
        padding=True
    ).to(device)

    # Forward pass with attention output
    with torch.no_grad():
        outputs = model(**inputs, output_attentions=True)

    # 마지막 레이어의 attention (평균)
    attention = outputs.attentions[-1]  # (batch, heads, seq, seq)
    attention = attention.mean(dim=1).squeeze()  # (seq, seq) - head 평균

    # [CLS] 토큰이 다른 토큰에 주는 attention
    cls_attention = attention[0, :].cpu().numpy()

    # 토큰 목록
    tokens = tokenizer.convert_ids_to_tokens(inputs['input_ids'].squeeze())

    return tokens, cls_attention

# Integrated Gradients 구현 (Captum 활용)
def interpret_with_ig(model, tokenizer, text, target_label, device):
    """
    Integrated Gradients를 사용한 토큰별 기여도 계산
    """
    model.eval()

    # 토큰화
    inputs = tokenizer(
        text,
        return_tensors='pt',
        truncation=True,
        max_length=256,
        padding=True
    ).to(device)

    input_ids = inputs['input_ids']
    attention_mask = inputs['attention_mask']

    # 임베딩 레이어 접근
    embeddings = model.bert.embeddings

    def forward_func(input_embeds):
        outputs = model(
            inputs_embeds=input_embeds,
            attention_mask=attention_mask
        )
        return outputs.logits[:, target_label]

    # Baseline (zero embedding)
    input_embeds = embeddings(input_ids)
    baseline = torch.zeros_like(input_embeds)

    # Integrated Gradients 계산
    ig = IntegratedGradients(forward_func)
    attributions = ig.attribute(
        input_embeds,
        baselines=baseline,
        n_steps=50
    )

    # 토큰별 기여도 (임베딩 차원 합산)
    token_attributions = attributions.sum(dim=-1).squeeze().cpu().detach().numpy()
    tokens = tokenizer.convert_ids_to_tokens(input_ids.squeeze())

    return tokens, token_attributions

# 다중 레이블 Stratified K-Fold (단순화를 위해 No Finding 기준)
def multilabel_kfold_validation(X, y, model_class, n_splits=5):
    """
    K-Fold 교차 검증
    """
    # No Finding 레이블 기준 stratification
    y_stratify = y[:, -1]  # No Finding 열

    skf = StratifiedKFold(n_splits=n_splits, shuffle=True, random_state=42)

    fold_results = []

    for fold, (train_idx, val_idx) in enumerate(skf.split(X, y_stratify)):
        X_train_fold, X_val_fold = X[train_idx], X[val_idx]
        y_train_fold, y_val_fold = y[train_idx], y[val_idx]

        # 모델 학습
        model = model_class()
        model.fit(X_train_fold, y_train_fold)

        # 예측 및 평가
        y_prob_fold = model.predict_proba(X_val_fold)

        # AUROC 계산
        aurocs = []
        for i in range(y.shape[1]):
            if len(np.unique(y_val_fold[:, i])) > 1:
                aurocs.append(roc_auc_score(y_val_fold[:, i], y_prob_fold[:, i]))

        mean_auroc = np.mean(aurocs) if aurocs else 0.0
        fold_results.append(mean_auroc)
        print(f"Fold {fold+1}: AUROC = {mean_auroc:.4f}")

    return fold_results
