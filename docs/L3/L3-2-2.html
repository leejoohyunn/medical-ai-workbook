def download_demo_data():
    """데모용 ABP 데이터 생성 - 실제 사용시 wfdb.rdsamp() 사용"""
    np.random.seed(42)

    # 시뮬레이션 ABP 데이터 생성
    fs = 125  # 샘플링 주파수
    duration = 3600  # 1시간 데이터
    n_samples = fs * duration
    n_patients = 100

    data = []
    for patient_id in range(n_patients):
        # 기본 ABP 파형 생성 (수축기/이완기 시뮬레이션)
        t = np.linspace(0, duration, n_samples)

        # 심박수 변동 시뮬레이션
        hr = 60 + 20 * np.sin(2 * np.pi * 0.001 * t) + np.random.randn(n_samples) * 2

        # ABP 파형 생성
        abp = 80 + 40 * np.sin(2 * np.pi * hr/60 * t) + np.random.randn(n_samples) * 5
        abp = signal.medfilt(abp, kernel_size=5)

        # 이벤트 레이블 생성 (20% positive)
        cv_event = 1 if np.random.random() < 0.2 else 0

        # 나이 및 동반질환
        age = np.random.randint(30, 90)
        comorbidities = np.random.randint(0, 5)

        data.append({
            'patient_id': f'P{patient_id:04d}',
            'abp_signal': abp,
            'cv_event': cv_event,
            'age': age,
            'comorbidities': comorbidities,
            'sampling_rate': fs
        })

    return pd.DataFrame(data)

def basic_preprocessing(signal_data, fs=125):
    """기본 전처리: 노이즈 제거 및 정규화"""
    # 1. 대역통과 필터 (0.5-40Hz)
    nyquist = fs / 2
    low = 0.5 / nyquist
    high = 40 / nyquist
    b, a = signal.butter(4, [low, high], btype='band')
    filtered = signal.filtfilt(b, a, signal_data)

    # 2. 정규화
    normalized = (filtered - np.mean(filtered)) / np.std(filtered)

    return normalized

import pywt

def advanced_preprocessing(signal_data, fs=125):
    """고급 전처리: 웨이블릿 노이즈 제거 + 적응형 필터"""

    # 1. 웨이블릿 노이즈 제거
    wavelet = 'db4'
    level = 6
    coeffs = pywt.wavedec(signal_data, wavelet, level=level)

    # 소프트 임계값 적용
    sigma = np.median(np.abs(coeffs[-1])) / 0.6745
    threshold = sigma * np.sqrt(2 * np.log(len(signal_data)))

    coeffs_thresh = list(coeffs)
    for i in range(1, len(coeffs)):
        coeffs_thresh[i] = pywt.threshold(coeffs[i], threshold, mode='soft')

    denoised = pywt.waverec(coeffs_thresh, wavelet)

    # 2. 적응형 필터 (RLS 근사)
    # 간단한 이동평균 기반 적응형 필터
    window_size = 25
    trend = pd.Series(denoised[:len(signal_data)]).rolling(window=window_size, center=True).mean()
    trend = trend.fillna(method='bfill').fillna(method='ffill')

    adaptive_filtered = denoised[:len(signal_data)] - trend * 0.3

    # 3. 최종 정규화
    final = (adaptive_filtered - np.mean(adaptive_filtered)) / np.std(adaptive_filtered)

    return final

def extract_bp_features(signal_data, fs=125):
    """혈압 파형에서 임상적 특징 추출"""

    features = {}

    # 1. 수축기/이완기 혈압 추정
    peaks, _ = signal.find_peaks(signal_data, distance=fs*0.5)  # 최소 0.5초 간격
    troughs, _ = signal.find_peaks(-signal_data, distance=fs*0.5)

    if len(peaks) > 0 and len(troughs) > 0:
        features['systolic_bp'] = np.mean(signal_data[peaks])
        features['diastolic_bp'] = np.mean(signal_data[troughs])
        features['pulse_pressure'] = features['systolic_bp'] - features['diastolic_bp']
        features['mean_arterial_pressure'] = (features['systolic_bp'] + 2*features['diastolic_bp']) / 3
    else:
        features['systolic_bp'] = np.max(signal_data)
        features['diastolic_bp'] = np.min(signal_data)
        features['pulse_pressure'] = features['systolic_bp'] - features['diastolic_bp']
        features['mean_arterial_pressure'] = np.mean(signal_data)

    # 2. 심박 변이성 특징
    if len(peaks) > 1:
        rr_intervals = np.diff(peaks) / fs * 1000  # ms 단위
        features['hrv_mean'] = np.mean(rr_intervals)
        features['hrv_std'] = np.std(rr_intervals)
        features['hrv_rmssd'] = np.sqrt(np.mean(np.diff(rr_intervals)**2))
    else:
        features['hrv_mean'] = 0
        features['hrv_std'] = 0
        features['hrv_rmssd'] = 0

    # 3. 주파수 도메인 특징
    freqs, psd = signal.welch(signal_data, fs=fs, nperseg=min(256, len(signal_data)//4))

    # LF (0.04-0.15 Hz) and HF (0.15-0.4 Hz) power
    lf_idx = np.where((freqs >= 0.04) & (freqs <= 0.15))
    hf_idx = np.where((freqs >= 0.15) & (freqs <= 0.4))

    features['lf_power'] = np.trapz(psd[lf_idx], freqs[lf_idx]) if len(lf_idx[0]) > 0 else 0
    features['hf_power'] = np.trapz(psd[hf_idx], freqs[hf_idx]) if len(hf_idx[0]) > 0 else 0
    features['lf_hf_ratio'] = features['lf_power'] / (features['hf_power'] + 1e-6)

    # 4. 복잡도 특징
    features['sample_entropy'] = entropy(np.histogram(signal_data, bins=20)[0] + 1e-6)

    return features

def domain_specific_preprocessing(signal_data, fs=125):
    """도메인 특화 전처리: 의학적 가이드라인 기반"""

    # 1. 생리학적 범위 필터링 (30-250 mmHg)
    clipped = np.clip(signal_data, -3, 3)  # 정규화된 신호 기준

    # 2. 아티팩트 제거 (급격한 변화)
    diff = np.diff(clipped)
    artifact_threshold = np.percentile(np.abs(diff), 99)
    artifact_idx = np.where(np.abs(diff) > artifact_threshold)[0]

    clean_signal = clipped.copy()
    for idx in artifact_idx:
        if idx > 0 and idx < len(clean_signal) - 1:
            clean_signal[idx] = (clean_signal[idx-1] + clean_signal[idx+1]) / 2

    # 3. 혈압 파형 특성 보존 필터
    # Savitzky-Golay 필터로 파형 형태 보존하며 평활화
    from scipy.signal import savgol_filter
    if len(clean_signal) > 51:
        smoothed = savgol_filter(clean_signal, window_length=51, polyorder=3)
    else:
        smoothed = clean_signal

    return smoothed

class LSTM_Model(nn.Module):
    """LSTM 기반 모델"""
    def __init__(self, input_size=1, hidden_size=128, num_layers=2, bidirectional=False):
        super(LSTM_Model, self).__init__()

        self.lstm = nn.LSTM(
            input_size=input_size,
            hidden_size=hidden_size,
            num_layers=num_layers,
            batch_first=True,
            dropout=0.2,
            bidirectional=bidirectional
        )

        lstm_out_size = hidden_size * 2 if bidirectional else hidden_size

        self.attention = nn.Linear(lstm_out_size, 1)
        self.fc1 = nn.Linear(lstm_out_size, 64)
        self.fc2 = nn.Linear(64, 1)
        self.dropout = nn.Dropout(0.5)

    def forward(self, x):
        # x shape: (batch, channels, seq_len) -> (batch, seq_len, channels)
        x = x.transpose(1, 2)

        lstm_out, _ = self.lstm(x)

        # Attention mechanism
        attention_weights = torch.softmax(self.attention(lstm_out), dim=1)
        attended = torch.sum(lstm_out * attention_weights, dim=1)

        x = self.dropout(attended)
        x = F.relu(self.fc1(x))
        x = torch.sigmoid(self.fc2(x))

        return x

class PositionalEncoding(nn.Module):
    """Transformer를 위한 위치 인코딩"""
    def __init__(self, d_model, max_len=5000):
        super(PositionalEncoding, self).__init__()

        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(torch.arange(0, d_model, 2).float() *
                           (-np.log(10000.0) / d_model))

        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)

        self.register_buffer('pe', pe.unsqueeze(0))

    def forward(self, x):
        return x + self.pe[:, :x.size(1)]

class TransformerModel(nn.Module):
    """Transformer 기반 시계열 분류 모델"""
    def __init__(self, input_dim=1, d_model=128, nhead=8, num_layers=4, seq_len=1250):
        super(TransformerModel, self).__init__()

        self.input_projection = nn.Linear(input_dim, d_model)
        self.positional_encoding = PositionalEncoding(d_model)

        encoder_layer = nn.TransformerEncoderLayer(
            d_model=d_model,
            nhead=nhead,
            dim_feedforward=512,
            dropout=0.1,
            batch_first=True
        )

        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)

        self.pooling = nn.AdaptiveAvgPool1d(1)
        self.fc1 = nn.Linear(d_model, 64)
        self.fc2 = nn.Linear(64, 1)
        self.dropout = nn.Dropout(0.5)

    def forward(self, x):
        # x shape: (batch, channels, seq_len) -> (batch, seq_len, channels)
        x = x.transpose(1, 2)

        # Project to d_model dimensions
        x = self.input_projection(x)
        x = self.positional_encoding(x)

        # Transformer encoding
        x = self.transformer(x)

        # Global pooling
        x = x.transpose(1, 2)
        x = self.pooling(x).squeeze(-1)

        # Classification head
        x = self.dropout(x)
        x = F.relu(self.fc1(x))
        x = torch.sigmoid(self.fc2(x))

        return x

def evaluate_model(model, test_loader, device='cpu'):
    """모델 평가 함수"""
    model.eval()
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for batch_x, batch_y in test_loader:
            batch_x = batch_x.to(device)
            outputs = model(batch_x)
            all_preds.extend(outputs.cpu().numpy())
            all_labels.extend(batch_y.numpy())

    all_preds = np.array(all_preds).squeeze()
    all_labels = np.array(all_labels).squeeze()

    # Calculate metrics
    auc = roc_auc_score(all_labels, all_preds)

    # Optimal threshold
    fpr, tpr, thresholds = roc_curve(all_labels, all_preds)
    optimal_idx = np.argmax(tpr - fpr)
    optimal_threshold = thresholds[optimal_idx]

    # Binary predictions
    binary_preds = (all_preds > optimal_threshold).astype(int)

    # Additional metrics
    from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score
    accuracy = accuracy_score(all_labels, binary_preds)
    precision = precision_score(all_labels, binary_preds)
    recall = recall_score(all_labels, binary_preds)
    f1 = f1_score(all_labels, binary_preds)

    return {
        'predictions': all_preds,
        'labels': all_labels,
        'auc': auc,
        'accuracy': accuracy,
        'precision': precision,
        'recall': recall,
        'f1': f1,
        'optimal_threshold': optimal_threshold,
        'fpr': fpr,
        'tpr': tpr
    }

def get_attention_weights(model, x, device='cpu'):
    """Transformer 모델의 attention weights 추출"""
    model.eval()
    x = x.to(device)

    # Hook을 사용하여 attention weights 추출
    attention_weights = []

    def hook_fn(module, input, output):
        if hasattr(module, 'self_attn'):
            attention_weights.append(output[1])

    # Register hook
    hooks = []
    for layer in model.transformer.layers:
        hooks.append(layer.register_forward_hook(hook_fn))

    with torch.no_grad():
        _ = model(x.unsqueeze(0))

    # Remove hooks
    for hook in hooks:
        hook.remove()

    return attention_weights
