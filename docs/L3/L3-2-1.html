def load_kaggle_eeg_subset(dataset_path, n_rows=3000):
    """
    EEG_Scaled_data.csv에서 앞쪽 n_rows 행만 읽어서 반환.
    dtype을 줄여서 RAM 사용량도 줄임.
    """
    # 1) CSV 파일 찾기
    csv_files = sorted(
        glob.glob(os.path.join(dataset_path, "**", "EEG_Scaled_data.csv"), recursive=True)
    )
    if not csv_files:
        raise FileNotFoundError("EEG_Scaled_data.csv를 찾지 못했습니다. path를 확인해주세요.")
    file_path = csv_files[0]
    print("📄 사용할 파일:", file_path)

    # 2) 헤더만 먼저 읽어서 컬럼 이름 확인
    header_df = pd.read_csv(file_path, nrows=0)
    cols = list(header_df.columns)

    if "target" not in cols:
        raise ValueError(f"'target' 컬럼을 찾지 못했습니다. cols = {cols}")

    # float / int dtype 설정해서 메모리 줄이기
    dtype = {c: np.float32 for c in cols if c != "target"}
    dtype["target"] = np.int8

    # 3) 앞에서부터 n_rows 행만 읽기
    df = pd.read_csv(
        file_path,
        usecols=cols,
        nrows=n_rows,
        dtype=dtype
    )

    print("🔢 읽은 행 수:", len(df))
    print("🔢 특성 수:", df.shape[1] - 1)  # target 제외
    print("🎯 라벨 분포:\n", df["target"].value_counts())

    return df

import numpy as np
import pandas as pd

def row_to_timeseries(df, row_idx=0, n_channels=18, fs=256):
    """
    df(예: df_small)에서 row_idx번째 샘플 1개를 꺼내
    (n_channels, n_samples) 형태의 시계열로 변환.

    반환:
    - eeg_data: (n_channels, n_samples)
    - labels: (n_samples,) 0/1 (세그먼트 전체에 동일 값)
    - time_axis: (n_samples,) 초 단위
    - seizure_segments: [(start_idx, end_idx), ...]
    - channel_names: ["Ch1", "Ch2", ...]
    - eeg_df: DataFrame (채널 + time + label)
    """
    feature_cols = [c for c in df.columns if c != "target"]
    n_features = len(feature_cols)

    # 1) reshape 가능 여부 확인
    if n_features % n_channels != 0:
        raise ValueError(
            f"n_channels={n_channels}로는 reshape 불가: "
            f"{n_features} % {n_channels} != 0\n"
            "→ n_channels를 16, 18, 32 등으로 바꿔보세요."
        )

    n_samples = n_features // n_channels
    duration_sec = n_samples / fs

    print(f"🧠 채널 수: {n_channels}")
    print(f"⏱ 샘플 수: {n_samples}, 길이: {duration_sec:.2f} 초")

    # 2) 해당 행의 feature / label 추출
    row = df.iloc[row_idx]
    segment_label = int(row["target"])
    feat = row[feature_cols].to_numpy(dtype=np.float32).flatten()

    # 3) reshape
    eeg_data = feat.reshape(n_channels, n_samples)

    # 4) 시간축 & 라벨 벡터
    time_axis = np.arange(n_samples) / fs
    labels = np.full(n_samples, segment_label, dtype=int)

    # 5) 발작 세그먼트 리스트
    seizure_segments = []
    if segment_label == 1:
        seizure_segments.append((0, n_samples))

    # 6) 채널 이름
    channel_names = [f"Ch{i+1}" for i in range(n_channels)]

    # 7) DataFrame 생성 (기존 구조 맞추기)
    eeg_df = pd.DataFrame(eeg_data.T, columns=channel_names)
    eeg_df["time"] = time_axis
    eeg_df["label"] = labels

    return eeg_data, labels, time_axis, seizure_segments, channel_names, eeg_df

class EEGPreprocessor:
    """EEG 신호 전처리 파이프라인"""

    def __init__(self, fs=256):
        self.fs = fs

    def bandpass_filter(self, data, lowcut=0.5, highcut=50):
        """대역통과 필터"""
        nyq = 0.5 * self.fs
        low = lowcut / nyq
        high = highcut / nyq
        b, a = signal.butter(4, [low, high], btype='band')
        return signal.filtfilt(b, a, data, axis=1)

    def notch_filter(self, data, freq=60):
        """노치 필터 (전력선 노이즈 제거)"""
        nyq = 0.5 * self.fs
        w0 = freq / nyq
        Q = 30.0  # Quality factor
        b, a = signal.iirnotch(w0, Q)
        return signal.filtfilt(b, a, data, axis=1)

    def normalize(self, data):
        """Z-score 정규화"""
        return (data - data.mean(axis=1, keepdims=True)) / (data.std(axis=1, keepdims=True) + 1e-8)

    def remove_artifacts_ica(self, data, n_components=20):
        """ICA를 이용한 아티팩트 제거 (시뮬레이션)"""
        # 실제로는 MNE의 ICA 사용
        # 여기서는 간단한 시뮬레이션
        return data * 0.9  # 아티팩트 10% 감소 시뮬레이션

def wavelet_denoise(signal_data, wavelet='db4', level=4):
    """웨이블릿 기반 노이즈 제거"""
    coeffs = pywt.wavedec(signal_data, wavelet, level=level)

    # 임계값 설정 (soft thresholding)
    threshold = 0.04
    coeffs_thresh = list(coeffs)
    coeffs_thresh[1:] = [pywt.threshold(c, threshold, mode='soft') for c in coeffs_thresh[1:]]

    # 재구성
    return pywt.waverec(coeffs_thresh, wavelet)

# 2.3 도메인 특화 특징 추출
from scipy import stats
from scipy.fft import fft, fftfreq

class EEGFeatureExtractor:
    """EEG 신호 특징 추출"""

    def __init__(self, fs=256):
        self.fs = fs

    def extract_time_features(self, signal_data):
        features = {
            'mean': np.mean(signal_data),
            'std': np.std(signal_data),
            'skewness': stats.skew(signal_data),
            'kurtosis': stats.kurtosis(signal_data),
            'rms': np.sqrt(np.mean(signal_data**2)),
            'zero_crossing': np.sum(np.diff(np.sign(signal_data)) != 0),
            'peak_to_peak': np.ptp(signal_data)
        }
        return features

    def extract_frequency_features(self, signal_data):
        # FFT (윈도우 길이에 맞춰 자동으로 길이 설정됨: 여기서는 512 샘플)
        fft_vals = np.abs(fft(signal_data))
        fft_freq = fftfreq(len(signal_data), 1/self.fs)

        # 양의 주파수만
        pos_freq_idx = fft_freq > 0
        fft_vals = fft_vals[pos_freq_idx]
        fft_freq = fft_freq[pos_freq_idx]

        # 대역별 파워
        delta_power = np.sum(fft_vals[(fft_freq >= 0.5) & (fft_freq < 4)]**2)
        theta_power = np.sum(fft_vals[(fft_freq >= 4) & (fft_freq < 8)]**2)
        alpha_power = np.sum(fft_vals[(fft_freq >= 8) & (fft_freq < 13)]**2)
        beta_power  = np.sum(fft_vals[(fft_freq >= 13) & (fft_freq < 30)]**2)
        gamma_power = np.sum(fft_vals[(fft_freq >= 30) & (fft_freq < 50)]**2)

        total_power = delta_power + theta_power + alpha_power + beta_power + gamma_power

        features = {
            'delta_power': delta_power,
            'theta_power': theta_power,
            'alpha_power': alpha_power,
            'beta_power': beta_power,
            'gamma_power': gamma_power,
            'delta_ratio': delta_power / total_power if total_power > 0 else 0,
            'theta_ratio': theta_power / total_power if total_power > 0 else 0,
            'alpha_ratio': alpha_power / total_power if total_power > 0 else 0,
            'beta_ratio': beta_power / total_power if total_power > 0 else 0,
            'gamma_ratio': gamma_power / total_power if total_power > 0 else 0,
            'spectral_entropy': -np.sum(fft_vals * np.log(fft_vals + 1e-10))
        }
        return features

    def extract_nonlinear_features(self, signal_data):
        diff1 = np.diff(signal_data)
        diff2 = np.diff(diff1)

        activity = np.var(signal_data)
        mobility = np.sqrt(np.var(diff1) / activity) if activity > 0 else 0
        complexity = np.sqrt(np.var(diff2) / np.var(diff1)) / mobility if mobility > 0 else 0

        features = {
            'hjorth_activity': activity,
            'hjorth_mobility': mobility,
            'hjorth_complexity': complexity,
            'approx_entropy': self._approx_entropy(signal_data, 2, 0.2 * np.std(signal_data))
        }
        return features

    def _approx_entropy(self, signal_data, m, r):
        N = len(signal_data)

        def _maxdist(xi, xj):
            return max(abs(ua - va) for ua, va in zip(xi, xj))

        def _phi(m):
            patterns = np.array([signal_data[i:i+m] for i in range(N-m+1)])
            C = [
                len([1 for j in range(len(patterns)) if _maxdist(patterns[i], patterns[j]) <= r])
                for i in range(len(patterns))
            ]
            return sum(np.log(c / (N-m+1)) for c in C) / (N-m+1)

        try:
            return _phi(m) - _phi(m+1)
        except Exception:
            return 0

def create_hybrid_model(input_shape, hp=None):
    """1D-CNN + BiLSTM 하이브리드 모델"""

    # 하이퍼파라미터 설정
    if hp is None:
        # 기본값
        conv1_filters = 64
        conv2_filters = 128
        lstm_units = 64
        dropout_rate = 0.3
        learning_rate = 0.001
    else:
        # Optuna 최적화용
        conv1_filters = hp.suggest_int('conv1_filters', 32, 128, step=32)
        conv2_filters = hp.suggest_int('conv2_filters', 64, 256, step=32)
        lstm_units = hp.suggest_int('lstm_units', 32, 128, step=32)
        dropout_rate = hp.suggest_float('dropout_rate', 0.2, 0.5)
        learning_rate = hp.suggest_loguniform('learning_rate', 1e-4, 1e-2)

    inputs = layers.Input(shape=input_shape)

    # CNN Branch
    x = layers.Conv1D(conv1_filters, kernel_size=5, activation='relu', padding='same')(inputs)
    x = layers.BatchNormalization()(x)
    x = layers.MaxPooling1D(pool_size=2)(x)
    x = layers.Dropout(dropout_rate)(x)

    x = layers.Conv1D(conv2_filters, kernel_size=3, activation='relu', padding='same')(x)
    x = layers.BatchNormalization()(x)
    x = layers.MaxPooling1D(pool_size=2)(x)
    x = layers.Dropout(dropout_rate)(x)

    # BiLSTM Branch
    x = layers.Bidirectional(layers.LSTM(lstm_units, return_sequences=True))(x)
    x = layers.Dropout(dropout_rate)(x)
    x = layers.Bidirectional(layers.LSTM(lstm_units//2))(x)

    # Output
    x = layers.Dense(32, activation='relu')(x)
    x = layers.Dropout(dropout_rate)(x)
    outputs = layers.Dense(1, activation='sigmoid')(x)

    model = Model(inputs, outputs)

    model.compile(
        optimizer=Adam(learning_rate=learning_rate),
        loss='binary_crossentropy',
        metrics=['accuracy', tf.keras.metrics.AUC(name='auc'),
                tf.keras.metrics.Precision(name='precision'),
                tf.keras.metrics.Recall(name='recall')]
    )

    return model

# 3.3 최종 모델 - Transformer 추가 및 앙상블

class TransformerBlock(layers.Layer):
    """Transformer 블록"""
    def __init__(self, embed_dim, num_heads, ff_dim, rate=0.1):
        super(TransformerBlock, self).__init__()
        self.att = layers.MultiHeadAttention(num_heads=num_heads, key_dim=embed_dim)
        self.ffn = keras.Sequential([
            layers.Dense(ff_dim, activation="relu"),
            layers.Dense(embed_dim),
        ])
        self.layernorm1 = layers.LayerNormalization(epsilon=1e-6)
        self.layernorm2 = layers.LayerNormalization(epsilon=1e-6)
        self.dropout1 = layers.Dropout(rate)
        self.dropout2 = layers.Dropout(rate)

    def call(self, inputs, training=None):
        # training 인자를 기본값 None으로 두는 게 Keras 관례
        attn_output = self.att(inputs, inputs)
        attn_output = self.dropout1(attn_output, training=training)
        out1 = self.layernorm1(inputs + attn_output)
        ffn_output = self.ffn(out1)
        ffn_output = self.dropout2(ffn_output, training=training)
        return self.layernorm2(out1 + ffn_output)


class PositionalEncoding(layers.Layer):
    """Sinusoidal Positional Encoding 레이어 (TF 연산은 여기 안에서만 사용)"""
    def __init__(self, embed_dim, **kwargs):
        super(PositionalEncoding, self).__init__(**kwargs)
        self.embed_dim = embed_dim

    def call(self, x):
        # x: (batch, time, embed_dim)
        seq_len = tf.shape(x)[1]

        # [0, 1, ..., seq_len-1]
        positions = tf.range(start=0, limit=seq_len, delta=1, dtype=tf.float32)  # (time,)

        # 각 차원별 주파수
        dims = tf.range(self.embed_dim, dtype=tf.float32)
        angle_rates = 1.0 / tf.pow(10000.0, (2 * (dims // 2)) / tf.cast(self.embed_dim, tf.float32))
        angle_rads = tf.expand_dims(positions, 1) * tf.expand_dims(angle_rates, 0)  # (time, embed_dim)

        # 짝수/홀수 인덱스에 sin/cos 적용
        sines = tf.sin(angle_rads[:, 0::2])
        cosines = tf.cos(angle_rads[:, 1::2])

        # embed_dim이 짝수일 때 기준으로 concat
        pe = tf.concat([sines, cosines], axis=-1)  # (time, embed_dim 또는 embed_dim-1)
        pe = pe[:, :self.embed_dim]               # 혹시 길이 안 맞으면 잘라서 맞춤
        pe = tf.expand_dims(pe, 0)                # (1, time, embed_dim)

        return x + pe  # broadcasting으로 (batch, time, embed_dim)에 더해짐


def create_transformer_model(input_shape):
    """Transformer 기반 모델"""
    inputs = layers.Input(shape=input_shape)   # (time_steps, 1)

    # Embedding 비슷한 역할: 특징 차원 64로 확장
    x = layers.Dense(64)(inputs)               # (batch, time_steps, 64)

    # Positional Encoding (커스텀 레이어 안에서만 tf 연산 사용)
    x = PositionalEncoding(embed_dim=64)(x)

    # Transformer blocks
    x = TransformerBlock(64, 4, 32)(x)
    x = TransformerBlock(64, 4, 32)(x)

    # Global pooling + 출력층
    x = layers.GlobalAveragePooling1D()(x)
    x = layers.Dropout(0.3)(x)
    x = layers.Dense(32, activation="relu")(x)
    x = layers.Dropout(0.3)(x)
    outputs = layers.Dense(1, activation="sigmoid")(x)

    model = Model(inputs, outputs)

    model.compile(
        optimizer=Adam(learning_rate=0.001),
        loss='binary_crossentropy',
        metrics=['accuracy', tf.keras.metrics.AUC(name='auc')]
    )

    return model

# 4. 성능 평가 - 긴 호흡

# 4.1 기본 메트릭
def evaluate_model(y_true, y_pred, y_pred_proba, model_name):
    """모델 성능 평가"""
    print(f"\n{'='*50}")
    print(f"{model_name} 성능 평가")
    print('='*50)

    # y_true에 실제로 어떤 클래스가 있는지 확인
    unique_classes = np.unique(y_true)

    # 기본 메트릭 (zero_division=0으로 안전하게)
    accuracy = accuracy_score(y_true, y_pred)
    precision = precision_score(y_true, y_pred, zero_division=0)
    recall = recall_score(y_true, y_pred, zero_division=0)
    f1 = f1_score(y_true, y_pred, zero_division=0)

    # AUC: 양쪽 클래스가 모두 있을 때만 계산
    if len(unique_classes) >= 2:
        try:
            auc = roc_auc_score(y_true, y_pred_proba)
        except ValueError:
            auc = np.nan
    else:
        auc = np.nan

    print(f"정확도 (Accuracy): {accuracy:.4f}")
    print(f"정밀도 (Precision): {precision:.4f}")
    print(f"재현율/민감도 (Recall/Sensitivity): {recall:.4f}")
    print(f"F1 Score: {f1:.4f}")
    if np.isnan(auc):
        print(f"AUC: nan (⚠️ y_true에 하나의 클래스만 존재)")
    else:
        print(f"AUC: {auc:.4f}")

    # 4.2 도메인 특화 메트릭
    # labels=[0,1]로 고정해서 항상 2x2 매트릭스를 얻음
    cm = confusion_matrix(y_true, y_pred, labels=[0, 1])
    print("\n📋 Confusion Matrix (labels=[0,1]):")
    print(cm)
    # 항상 2x2이므로 ravel() 가능 (빈 클래스면 해당 행/열이 전부 0)
    tn, fp, fn, tp = cm.ravel()

    specificity = tn / (tn + fp) if (tn + fp) > 0 else 0.0
    fpr = fp / (fp + tn) if (fp + tn) > 0 else 0.0  # False Positive Rate
    fnr = fn / (fn + tp) if (fn + tp) > 0 else 0.0  # False Negative Rate

    print(f"\n특이도 (Specificity): {specificity:.4f}")
    print(f"오경보율 (False Positive Rate): {fpr:.4f}")
    print(f"미검출율 (False Negative Rate): {fnr:.4f}")

    # 4.3 임상적 메트릭
    # Positive Predictive Value (PPV)
    ppv = tp / (tp + fp) if (tp + fp) > 0 else 0.0
    # Negative Predictive Value (NPV)
    npv = tn / (tn + fn) if (tn + fn) > 0 else 0.0

    print(f"\n양성 예측도 (PPV): {ppv:.4f}")
    print(f"음성 예측도 (NPV): {npv:.4f}")

    # Detection Delay (시뮬레이션)
    detection_delay = np.random.uniform(5, 15)  # 분 단위
    print(f"평균 검출 지연시간: {detection_delay:.1f}분")

    if len(unique_classes) < 2:
        print("⚠️ y_true에 한 클래스만 존재합니다. 일부 메트릭(AUC 등) 해석에 주의가 필요합니다.")

    return {
        'accuracy': accuracy,
        'precision': precision,
        'recall': recall,
        'f1': f1,
        'auc': auc,
        'specificity': specificity,
        'fpr': fpr
    }

# 특징 중요도 분석
def analyze_feature_importance(model, X_sample, feature_names=None):
    """Gradient 기반 특징 중요도 분석"""
    # GradientTape를 사용한 특징 중요도 계산
    X_tensor = tf.convert_to_tensor(X_sample, dtype=tf.float32)

    with tf.GradientTape() as tape:
        tape.watch(X_tensor)
        predictions = model(X_tensor)

    gradients = tape.gradient(predictions, X_tensor)
    importance = tf.reduce_mean(tf.abs(gradients), axis=0).numpy()

    return importance
