# Finding Labels를 원-핫 인코딩으로 변환
def create_label_columns(df_raw, disease_labels):
    """Finding Labels 문자열을 각 질환별 이진 컬럼으로 변환"""
    df = df_raw.copy()

    for disease in disease_labels:
        df[disease] = df['Finding Labels'].apply(lambda x: 1 if disease in x else 0)

    return df

2.7.3 1.3 환자 단위 데이터 분할

🔍 학습 포인트 #2
의료 영상에서는 환자 단위 분할(Patient-level split)이 필수입니다. 동일 환자의 여러 이미지가 학습과 테스트 세트에 나뉘면 데이터 누출이 발생합니다.

Code 
def patient_level_split(df, test_size=0.15, val_size=0.15, random_state=42):
    unique_patients = df['Patient ID'].unique()

    train_val_patients, test_patients = train_test_split(
        unique_patients, test_size=test_size, random_state=random_state
    )
    train_patients, val_patients = train_test_split(
        train_val_patients, test_size=val_size/(1-test_size), random_state=random_state
    )

# MixUp 구현
class MixUp:
    def __init__(self, alpha=0.4):
        self.alpha = alpha

    def __call__(self, images, labels):
        lam = np.random.beta(self.alpha, self.alpha)
        indices = torch.randperm(images.size(0)).to(images.device)
        mixed_images = lam * images + (1 - lam) * images[indices]
        mixed_labels = lam * labels + (1 - lam) * labels[indices]
        return mixed_images, mixed_labels

# CutMix 구현
class CutMix:
    def __init__(self, alpha=1.0):
        self.alpha = alpha

    def rand_bbox(self, size, lam):
        W, H = size[2], size[3]
        cut_rat = np.sqrt(1. - lam)
        cut_w, cut_h = int(W * cut_rat), int(H * cut_rat)
        cx, cy = np.random.randint(W), np.random.randint(H)
        return (np.clip(cx - cut_w//2, 0, W), np.clip(cy - cut_h//2, 0, H),
                np.clip(cx + cut_w//2, 0, W), np.clip(cy + cut_h//2, 0, H))

    def __call__(self, images, labels):
        lam = np.random.beta(self.alpha, self.alpha)
        indices = torch.randperm(images.size(0)).to(images.device)
        bbx1, bby1, bbx2, bby2 = self.rand_bbox(images.size(), lam)

        mixed_images = images.clone()
        mixed_images[:, :, bbx1:bbx2, bby1:bby2] = images[indices, :, bbx1:bbx2, bby1:bby2]
        lam = 1 - ((bbx2 - bbx1) * (bby2 - bby1) / (images.size(-1) * images.size(-2)))
        mixed_labels = lam * labels + (1 - lam) * labels[indices]
        return mixed_images, mixed_labels

def compute_class_weights(df, label_cols):
    pos_counts = df[label_cols].sum()
    neg_counts = len(df) - pos_counts
    pos_weights = np.clip((neg_counts / (pos_counts + 1e-5)).values, 1, 50)
    return torch.tensor(pos_weights, dtype=torch.float32)

# 학습 및 평가 함수
def train_one_epoch(model, train_loader, criterion, optimizer, scaler, device, use_mixup=False, mixup_fn=None):
    model.train()
    running_loss = 0.0

    for images, labels in tqdm(train_loader, desc='Training'):
        images, labels = images.to(device), labels.to(device)

        if use_mixup and mixup_fn and np.random.random() < 0.5:
            images, labels = mixup_fn(images, labels)

        optimizer.zero_grad()
        with autocast():
            outputs = model(images)
            loss = criterion(outputs, labels)

        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()
        running_loss += loss.item()

    return running_loss / len(train_loader)

@torch.no_grad()
def evaluate(model, data_loader, criterion, device):
    model.eval()
    running_loss = 0.0
    all_preds, all_labels = [], []

    for images, labels in tqdm(data_loader, desc='Evaluating'):
        images, labels = images.to(device), labels.to(device)
        outputs = model(images)
        running_loss += criterion(outputs, labels).item()
        all_preds.append(torch.sigmoid(outputs).cpu().numpy())
        all_labels.append(labels.cpu().numpy())

    all_preds, all_labels = np.vstack(all_preds), np.vstack(all_labels)
    aurocs = [roc_auc_score(all_labels[:, i], all_preds[:, i]) if all_labels[:, i].sum() > 0 else 0.5
              for i in range(all_labels.shape[1])]

    return running_loss / len(data_loader), np.mean(aurocs), aurocs, all_preds, all_labels

class ChestXrayResNet(nn.Module):
    def __init__(self, num_classes=14, pretrained=True, freeze_backbone=True):
        super().__init__()
        self.backbone = models.resnet50(pretrained=pretrained)




        if freeze_backbone:
            for param in self.backbone.parameters():
                param.requires_grad = False

        in_features = self.backbone.fc.in_features
        self.backbone.fc = nn.Sequential(
            nn.Linear(in_features, 512), nn.ReLU(), nn.Dropout(0.5),
            nn.Linear(512, num_classes)
        )

    def forward(self, x):
        return self.backbone(x)

    def unfreeze_backbone(self, num_layers=None):
        if num_layers is None:
            for param in self.backbone.parameters():
                param.requires_grad = True
        else:
            for layer in list(self.backbone.children())[-num_layers:]:
                for param in layer.parameters():
                    param.requires_grad = True

def segment_with_sam(predictor, image, point_coords, point_labels):
    """SAM 포인트 프롬프트 기반 세그멘테이션"""
    predictor.set_image(image)
    masks, scores, _ = predictor.predict(
        point_coords=point_coords,
        point_labels=point_labels,
        multimask_output=True
    )
    return masks[np.argmax(scores)]

def gradcam_to_sam_prompt(heatmap, threshold=0.5):
    """Grad-CAM에서 SAM 프롬프트 포인트 추출"""
    from scipy import ndimage
    binary = (heatmap > threshold).astype(np.uint8)
    labeled, n = ndimage.label(binary)

    points = []
    for i in range(1, n + 1):
        y, x = np.where(labeled == i)
        if len(y) > 0:
            points.append([int(np.mean(x)), int(np.mean(y))])

    return np.array(points[:1]) if points else None

def test_time_augmentation(model, image_tensor, device, n_augments=5):
    """Test-Time Augmentation"""
    model.eval()

    transforms = [
        lambda x: x,
        lambda x: torch.flip(x, dims=[-1]),
        lambda x: torch.rot90(x, k=1, dims=[-2, -1]),
        lambda x: torch.rot90(x, k=3, dims=[-2, -1]),
        lambda x: x + torch.randn_like(x) * 0.05,
    ][:n_augments]

    predictions = []
    with torch.no_grad():
        for t in transforms:
            output = model(t(image_tensor.clone()).unsqueeze(0).to(device))
            predictions.append(torch.sigmoid(output).cpu().numpy())

    return np.mean(predictions, axis=0)
