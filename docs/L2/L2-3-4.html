2 🧬 L2-3 의료 영상 분석: 피부 병변 흑색종 분류

2.1 🧭 학습 목표 (Learning Outcomes)

이 워크북을 통해 학습자는 다음을 실습하게 됩니다:

  • 피부 병변 이미지를 로드하고 기본 탐색적 분석(EDA)을 수행할 수 있다
  • 다양한 이미지 전처리 기법(정규화, 히스토그램 평활화, CLAHE)을 적용하고 효과를 비교할 수 있다
  • 전통적 특징 추출 기법(픽셀 통계, HOG, SIFT)을 활용하여 이미지 특징을 벡터화할 수 있다
  • SVM 기반 분류 모델을 단계적으로 개선(Linear → RBF → SMOTE + 최적화)할 수 있다
  • SHAP을 활용하여 모델 예측을 해석하고 임상적 의미를 도출할 수 있다

2.2 🧩 실습 개요

  • 데이터: ISIC Skin Lesion Dataset (Kaggle)
  • 목표: 피부 병변 이미지에서 흑색종(Melanoma)과 양성 병변을 분류하는 분류기 구축
  • 사용 기술: OpenCV, scikit-image, scikit-learn, SHAP
  • 제약 조건:
    1. 딥러닝 없이 전통적 머신러닝(SVM) 기반으로 분류
    2. 클래스 불균형 문제를 SMOTE로 해결
    3. 하이퍼파라미터 튜닝을 통한 성능 최적화 필수

def load_image(image_id):
    """이미지 ID로 이미지를 로드하는 함수"""
    # 두 폴더 중 하나에서 이미지 찾기
    img_path_1 = IMAGE_DIR_1 / f"{image_id}.jpg"
    img_path_2 = IMAGE_DIR_2 / f"{image_id}.jpg"
    
    if img_path_1.exists():
        img = cv2.imread(str(img_path_1))
    elif img_path_2.exists():
        img = cv2.imread(str(img_path_2))
    else:
        return None
    
    # BGR to RGB 변환
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    return img

def preprocess_histogram_eq(img):
    """
    히스토그램 평활화를 적용한 전처리
    - LAB 색 공간에서 L 채널에 평활화 적용
    """
    # 크기 조정
    img_resized = cv2.resize(img, TARGET_SIZE)
    
    # LAB 색 공간 변환
    lab = cv2.cvtColor(img_resized, cv2.COLOR_RGB2LAB)
    
    # L 채널에 히스토그램 평활화 적용
    lab[:, :, 0] = cv2.equalizeHist(lab[:, :, 0])
    
    # RGB로 다시 변환
    img_eq = cv2.cvtColor(lab, cv2.COLOR_LAB2RGB)
    
    # 정규화
    img_normalized = img_eq.astype(np.float32) / 255.0
    
    return img_normalized

def preprocess_clahe(img, clip_limit=2.0, tile_grid_size=(8, 8)):
    """
    CLAHE를 적용한 고급 전처리
    - 지역적 적응형 히스토그램 평활화
    - clip_limit으로 노이즈 증폭 방지
    """
    # 크기 조정
    img_resized = cv2.resize(img, TARGET_SIZE)
    
    # LAB 색 공간 변환
    lab = cv2.cvtColor(img_resized, cv2.COLOR_RGB2LAB)
    
    # CLAHE 객체 생성
    clahe = cv2.createCLAHE(clipLimit=clip_limit, tileGridSize=tile_grid_size)
    
    # L 채널에 CLAHE 적용
    lab[:, :, 0] = clahe.apply(lab[:, :, 0])
    
    # RGB로 다시 변환
    img_clahe = cv2.cvtColor(lab, cv2.COLOR_LAB2RGB)
    
    # 정규화
    img_normalized = img_clahe.astype(np.float32) / 255.0
    
    return img_normalized

# 테스트
clahe_processed = preprocess_clahe(sample_img)

def extract_color_histogram(img, bins=32):
    """
    색상 히스토그램 특징 추출
    - RGB 및 HSV 색 공간에서 히스토그램 추출
    """
    features = []
    
    # RGB 히스토그램
    for channel in range(3):
        hist, _ = np.histogram(img[:, :, channel], bins=bins, range=(0, 1))
        hist = hist.astype(np.float32) / hist.sum()  # 정규화
        features.extend(hist)
    
    # HSV로 변환
    hsv = cv2.cvtColor((img * 255).astype(np.uint8), cv2.COLOR_RGB2HSV)
    hsv = hsv.astype(np.float32) / 255.0
    
    # HSV 히스토그램
    for channel in range(3):
        hist, _ = np.histogram(hsv[:, :, channel], bins=bins, range=(0, 1))
        hist = hist.astype(np.float32) / hist.sum()
        features.extend(hist)
    
    return np.array(features)

def predict_melanoma(image_path, model_path=MODEL_PATH, scaler_path=SCALER_PATH):
    """
    새로운 이미지에 대한 흑색종 예측 함수
    
    Parameters:
    -----------
    image_path : str
        예측할 이미지 파일 경로
    
    Returns:
    --------
    dict : 예측 결과 (클래스, 확률)
    """
    # 모델 및 스케일러 로드
    model = joblib.load(model_path)
    scaler = joblib.load(scaler_path)
    
    # 이미지 로드
    img = cv2.imread(image_path)
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    
    # 특징 추출
    features = extract_all_features(img)
    features_scaled = scaler.transform(features.reshape(1, -1))
    
    # 예측
    prediction = model.predict(features_scaled)[0]
    probability = model.predict_proba(features_scaled)[0]
    
    result = {
        'prediction': 'Melanoma' if prediction == 1 else 'Non-Melanoma',
        'confidence': max(probability),
        'melanoma_probability': probability[1]
    }
    
    return result
