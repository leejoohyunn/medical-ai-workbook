def aggregate_diagnostic(y_dict):
    """SCP 코드를 슈퍼클래스로 매핑"""

def load_ecg_signal(filename, sampling_rate=100):
    """ECG 신호 로드 함수"""
    if sampling_rate == 100:
        record = wfdb.rdsamp(DATA_PATH + filename)
    else:
        record = wfdb.rdsamp(DATA_PATH + filename.replace('_lr', '_hr'))
    return record[0]  # (5000, 12) for 100Hz or (50000, 12) for 500Hz

🔍 학습 포인트 #1: 멀티모달 데이터의 특성 이해

PTB-XL 데이터셋은 두 가지 모달리티를 제공합니다: 1. 파형 데이터 (시계열): 12-리드 ECG, 1000 timesteps (10초 @ 100Hz) 2. 메타데이터 (정형): 환자 인구통계학적 정보 (나이, 성별, 신장, 체중)

멀티모달 분석의 핵심은 각 모달리티가 가진 상보적 정보를 효과적으로 융합하는 것입니다: - ECG 파형: 심장의 전기적 활동 패턴 (동적 정보) - 환자 속성: 질환 위험 요인과 관련된 배경 정보 (정적 정보)

def preprocess_ecg(signal_data, fs=100):
    """
    ECG 신호 전처리 파이프라인
    1. 기저선 변동 제거 (고역통과 필터)
    2. 고주파 노이즈 제거 (저역통과 필터)
    3. 전력선 노이즈 제거 (노치 필터)
    """
    processed = np.zeros_like(signal_data)
    
    for lead_idx in range(signal_data.shape[1]):
        lead_signal = signal_data[:, lead_idx]
        
        # 1. 기저선 제거 (0.5Hz 고역통과)
        b, a = signal.butter(2, 0.5/(fs/2), btype='high')
        lead_signal = signal.filtfilt(b, a, lead_signal)
        
        # 2. 고주파 노이즈 제거 (40Hz 저역통과)
        b, a = signal.butter(4, 40/(fs/2), btype='low')
        lead_signal = signal.filtfilt(b, a, lead_signal)
        
        # 3. 전력선 노이즈 제거 (50Hz 노치)
        b, a = signal.iirnotch(50, 30, fs)
        lead_signal = signal.filtfilt(b, a, lead_signal)
        
        processed[:, lead_idx] = lead_signal
    
    return processed

def extract_ecg_features(signal_data, fs=100):
    """
    ECG 신호에서 다양한 특징 추출
    1. 통계적 특징
    2. R-peak 기반 특징
    3. HRV 특징 (시간/주파수 도메인)
    """
    features = {}
    
    # Lead II 사용 (가장 표준적인 리드)
    lead_ii = signal_data[:, 1]
    
    # 1. 통계적 특징 (12리드 전체)
    for i, lead_name in enumerate(['I', 'II', 'III', 'aVR', 'aVL', 'aVF', 'V1', 'V2', 'V3', 'V4', 'V5', 'V6']):
        lead = signal_data[:, i]
        features[f'{lead_name}_mean'] = np.mean(lead)
        features[f'{lead_name}_std'] = np.std(lead)
        features[f'{lead_name}_max'] = np.max(lead)
        features[f'{lead_name}_min'] = np.min(lead)
        features[f'{lead_name}_skew'] = skew(lead)
        features[f'{lead_name}_kurtosis'] = kurtosis(lead)
    
    # 2. R-peak 및 HRV 특징 (NeuroKit2 사용)
    try:
        # R-peak 검출
        _, rpeaks = nk.ecg_peaks(lead_ii, sampling_rate=fs)
        r_peaks = rpeaks['ECG_R_Peaks']
        
        if len(r_peaks) > 2:
            # RR 간격 계산
            rr_intervals = np.diff(r_peaks) / fs * 1000  # ms 단위
            
            # 시간 도메인 HRV
            features['hr_mean'] = 60000 / np.mean(rr_intervals)  # BPM
            features['hr_std'] = np.std(60000 / rr_intervals)
            features['rmssd'] = np.sqrt(np.mean(np.diff(rr_intervals)**2))
            features['sdnn'] = np.std(rr_intervals)
            features['nn50'] = np.sum(np.abs(np.diff(rr_intervals)) > 50)
            features['pnn50'] = features['nn50'] / len(rr_intervals) * 100
            
            # 주파수 도메인 HRV (간략화)
            features['rr_mean'] = np.mean(rr_intervals)
            features['rr_range'] = np.max(rr_intervals) - np.min(rr_intervals)
        else:
            # R-peak 검출 실패 시 기본값
            for hrv_feat in ['hr_mean', 'hr_std', 'rmssd', 'sdnn', 'nn50', 'pnn50', 'rr_mean', 'rr_range']:
                features[hrv_feat] = np.nan
                
    except Exception as e:
        for hrv_feat in ['hr_mean', 'hr_std', 'rmssd', 'sdnn', 'nn50', 'pnn50', 'rr_mean', 'rr_range']:
            features[hrv_feat] = np.nan
    
    return features
