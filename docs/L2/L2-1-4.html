2.1 🧭 학습 목표 (Learning Outcomes)

이 워크북을 통해 학습자는 다음을 실습하게 됩니다:

  • 응급실 환자 데이터의 복잡한 결측치 패턴을 이해하고 MICE를 활용한 고급 대체 기법 적용
  • 심각한 클래스 불균형 문제를 ADASYN 등 다양한 방법으로 해결하며 성능 변화 관찰
  • Baseline 모델부터 앙상블까지 ’긴 호흡’으로 성능을 점진적으로 개선
  • SHAP을 활용한 모델 해석으로 응급실 재방문 리스크 요인 파악

2.2 🧩 실습 개요

  • 데이터: Emergency Department Visits Dataset
  • 목표: 72시간 내 응급실 재방문 고위험 환자 예측
  • 사용 기술: pandas, scikit-learn, imbalanced-learn (ADASYN), SHAP, Optuna
  • 제약 조건:
    1. 재방문 예측에서 False Negative를 최소화 (놓치는 환자 최소화)
    2. 극심한 클래스 불균형 (재방문율 ~5%) 효과적 처리

np.random.seed(42)
n_samples = 10000

n_revisit = 500
n_no_revisit = 9500

# 특징 생성
def generate_er_data():
    data = []

    # 재방문 환자 (고위험군)
    for _ in range(n_revisit):
        record = {
            'age': np.random.normal(75, 15),  # 고령
            'gender': np.random.choice([0, 1]),
            'triage_level': np.random.choice([1, 2, 3], p=[0.3, 0.5, 0.2]),  # 높은 중증도
            'ed_los': np.random.exponential(8),  # 긴 체류시간
            'num_procedures': np.random.poisson(4),
            'num_medications': np.random.poisson(6),
            'previous_visits': np.random.poisson(3),  # 많은 과거 방문
            'systolic_bp': np.random.normal(140, 25),
            'heart_rate': np.random.normal(95, 20),
            'respiratory_rate': np.random.normal(22, 5),
            'temperature': np.random.normal(37.5, 1.2),
            'oxygen_saturation': np.random.normal(94, 4),
            'pain_score': np.random.randint(4, 11),
            'discharge_home': 0.3,  # 낮은 귀가율
            'revisit_72hr': 1
        }
        data.append(record)

    # 재방문하지 않은 환자
    for _ in range(n_no_revisit):
        record = {
            'age': np.random.normal(55, 20),
            'gender': np.random.choice([0, 1]),
            'triage_level': np.random.choice([2, 3, 4, 5], p=[0.2, 0.3, 0.3, 0.2]),
            'ed_los': np.random.exponential(4),
            'num_procedures': np.random.poisson(2),
            'num_medications': np.random.poisson(3),
            'previous_visits': np.random.poisson(1),
            'systolic_bp': np.random.normal(125, 20),
            'heart_rate': np.random.normal(80, 15),
            'respiratory_rate': np.random.normal(18, 3),
            'temperature': np.random.normal(37.0, 0.5),
            'oxygen_saturation': np.random.normal(97, 2),
            'pain_score': np.random.randint(1, 8),
            'discharge_home': 0.8,
            'revisit_72hr': 0
        }
        data.append(record)

    df = pd.DataFrame(data)

    # 나이 제한 및 정수 변환
    df['age'] = df['age'].clip(18, 100).astype(int)

    # 결측치 생성 (현실적인 패턴)
    missing_cols = ['oxygen_saturation', 'pain_score', 'temperature']
    for col in missing_cols:
        mask = np.random.random(len(df)) < 0.1  # 10% 결측
        df.loc[mask, col] = np.nan

    return df

    df_new = df.copy()

    # 1. 나이 그룹 (임상적 의미)

                                  bins=[0, 18, 45, 65, 100],
                                  labels=['young', 'adult', 'senior', 'elderly'])
    df_new['age_group'] = df_new['age_group'].cat.codes

    # 2. 활력징후 위험 점수
    df_new['vital_risk_score'] = 0
    df_new.loc[df['systolic_bp'] > 140, 'vital_risk_score'] += 1
    df_new.loc[df['systolic_bp'] < 90, 'vital_risk_score'] += 2
    df_new.loc[df['heart_rate'] > 100, 'vital_risk_score'] += 1
    df_new.loc[df['heart_rate'] < 60, 'vital_risk_score'] += 1
    df_new.loc[df['respiratory_rate'] > 20, 'vital_risk_score'] += 1
    df_new.loc[df['oxygen_saturation'] < 95, 'vital_risk_score'] += 2

    # 3. 복잡도 점수
    df_new['complexity_score'] = (
        df['num_procedures'] +
        df['num_medications'] / 2 +
        df['ed_los'] / 4
    )

    # 4. 고위험 환자 플래그
    df_new['high_risk_flag'] = (
        (df['age'] > 70) &
        (df['previous_visits'] > 2) &
        (df['triage_level'] <= 2)

    return df_new
