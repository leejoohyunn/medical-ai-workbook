# 샘플 데이터 생성 (실제 환경에서는 Kaggle에서 다운로드)
# 시뮬레이션을 위한 코드입니다
np.random.seed(42)

def create_sample_histopathology_data(n_samples=1000):
    """유방암 조직 병리 이미지 샘플 데이터 생성"""

    images = []
    labels = []

    for i in range(n_samples):
        # 양성(0) 또는 악성(1) 레이블 생성
        label = np.random.choice([0, 1], p=[0.7, 0.3])  # 70% 양성, 30% 악성

        # H&E 염색 이미지 시뮬레이션 (50x50 RGB)
        if label == 0:  # 양성
            # 보라색-분홍색 톤 (정상 조직)
            base_color = np.array([200, 150, 200])
            noise_level = 20
        else:  # 악성
            # 진한 보라색 톤 (암 조직 - 핵 밀도 증가)
            base_color = np.array([150, 100, 180])
            noise_level = 30

        # 이미지 생성
        img = np.zeros((50, 50, 3), dtype=np.uint8)
        for c in range(3):
            img[:, :, c] = base_color[c] + np.random.randn(50, 50) * noise_level

        # 세포핵 시뮬레이션 추가
        n_nuclei = np.random.randint(5, 15) if label == 0 else np.random.randint(10, 25)
        for _ in range(n_nuclei):
            center = (np.random.randint(5, 45), np.random.randint(5, 45))
            radius = np.random.randint(2, 5)
            cv2.circle(img, center, radius, (100, 50, 150), -1)

        img = np.clip(img, 0, 255).astype(np.uint8)
        images.append(img)
        labels.append(label)

    return np.array(images), np.array(labels)

# 데이터 생성
X_images, y_labels = create_sample_histopathology_data(1000)

print(f"📊 데이터셋 크기: {X_images.shape}")

print(f"   - 악성 (1): {np.sum(y_labels == 1)} ({np.sum(y_labels == 1)/len(y_labels)*100:.1f}%)")

### 2.1 기본 전처리

def basic_preprocessing(image):
    """기본 전처리: 정규화 및 리사이즈"""
    # 그레이스케일 변환
    gray = cv2.cvtColor(image, cv2.COLOR_RGB2GRAY)

    # 0-1 정규화
    normalized = gray.astype(np.float32) / 255.0

    return normalized

def histogram_equalization(image):
    """히스토그램 평활화"""
    gray = cv2.cvtColor(image, cv2.COLOR_RGB2GRAY)
    equalized = cv2.equalizeHist(gray)
    return equalized.astype(np.float32) / 255.0

def apply_clahe(image, clip_limit=2.0, grid_size=(8, 8)):
    """CLAHE 적용"""
    gray = cv2.cvtColor(image, cv2.COLOR_RGB2GRAY)
    clahe = cv2.createCLAHE(clipLimit=clip_limit, tileGridSize=grid_size)
    enhanced = clahe.apply(gray)
    return enhanced.astype(np.float32) / 255.0

def color_deconvolution_simulation(image):
    """
    색상 분해 시뮬레이션 (H&E 염색 분리)
    실제로는 더 복잡한 알고리즘이 필요하지만, 여기서는 간단히 시뮬레이션
    """
    # H&E 염색의 대략적인 색상 채널 분리
    hematoxylin = image[:, :, 2]  # 파란색 채널 (핵)
    eosin = image[:, :, 0]  # 빨간색 채널 (세포질)

    # 두 염색의 조합으로 최종 이미지 생성
    combined = 0.7 * hematoxylin + 0.3 * eosin
    combined = np.clip(combined, 0, 255).astype(np.uint8)

    return combined.astype(np.float32) / 255.0

def morphological_processing(image):
    """형태학적 처리로 노이즈 제거"""
    gray = cv2.cvtColor(image, cv2.COLOR_RGB2GRAY)

    # 모폴로지 연산으로 노이즈 제거
    kernel = np.ones((3, 3), np.uint8)
    denoised = cv2.morphologyEx(gray, cv2.MORPH_OPEN, kernel)
    denoised = cv2.morphologyEx(denoised, cv2.MORPH_CLOSE, kernel)

    return denoised.astype(np.float32) / 255.0

def extract_pixel_statistics(image):
    """픽셀 통계 특징 추출"""
    features = [
        np.mean(image),
        np.std(image),
        np.min(image),
        np.max(image),
        np.median(image)
    ]
    return np.array(features)

def extract_hog_features(image, orientations=9, pixels_per_cell=(8, 8),
                        cells_per_block=(2, 2)):
    """HOG (Histogram of Oriented Gradients) 특징 추출"""
    features = hog(image, orientations=orientations,
                  pixels_per_cell=pixels_per_cell,
                  cells_per_block=cells_per_block,
                  visualize=False,
                  feature_vector=True)
    return features

def extract_lbp_features(image, P=24, R=3):
    """LBP (Local Binary Pattern) 특징 추출"""
    lbp = local_binary_pattern(image, P, R, method='uniform')
    hist, _ = np.histogram(lbp.ravel(), bins=np.arange(0, P + 3),
                          range=(0, P + 2))
    hist = hist.astype("float")
    hist /= (hist.sum() + 1e-7)  # 정규화
    return hist

def extract_combined_features(image):
    """모든 특징을 결합"""
    stats = extract_pixel_statistics(image)
    hog_feat = extract_hog_features(image)
    lbp_feat = extract_lbp_features(image)

    return np.concatenate([stats, hog_feat, lbp_feat])
