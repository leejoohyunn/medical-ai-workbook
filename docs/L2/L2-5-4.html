# 데이터 품질 검증 함수
def validate_data_quality(df, name):
    """데이터프레임의 품질을 검증하는 함수"""
    print(f"\n{'='*50}")
    print(f"📊 {name} 데이터 품질 검증")
    print(f"{'='*50}")

    # 기본 정보
    print(f"\n📏 데이터 크기: {df.shape[0]:,} rows × {df.shape[1]} columns")

    # 결측값 분석
    missing = df.isnull().sum()
    missing_pct = (missing / len(df) * 100).round(2)
    missing_df = pd.DataFrame({
        'Missing Count': missing,
        'Missing %': missing_pct
    })
    missing_df = missing_df[missing_df['Missing Count'] > 0]

    if len(missing_df) > 0:
        print(f"\n⚠️ 결측값 현황:")
        display(missing_df)
    else:
        print(f"\n✅ 결측값 없음")

    # 중복값 확인
    duplicates = df.duplicated().sum()
    print(f"\n🔄 중복 행 수: {duplicates:,}")

    return missing_df

# DICOM 파일 경로 설정
import pydicom
import glob

# 이미지 파일 경로 찾기
image_dir = '/root/.cache/kagglehub/competitions/siim-covid19-detection/train'
dicom_files = glob.glob(f'{image_dir}/**/*.dcm', recursive=True)

print(f"📁 총 DICOM 파일 수: {len(dicom_files):,}")

# 샘플 이미지 로드 및 메타데이터 추출
def load_dicom_info(file_path):
    """DICOM 파일에서 이미지와 메타데이터 추출"""
    dcm = pydicom.dcmread(file_path)

    # 이미지 추출
    image = dcm.pixel_array

    # 메타데이터 추출
    metadata = {
        'PatientSex': getattr(dcm, 'PatientSex', 'Unknown'),
        'PatientAge': getattr(dcm, 'PatientAge', 'Unknown'),
        'ViewPosition': getattr(dcm, 'ViewPosition', 'Unknown'),
        'Rows': dcm.Rows,
        'Columns': dcm.Columns,
        'PhotometricInterpretation': getattr(dcm, 'PhotometricInterpretation', 'Unknown')
    }

    return image, metadata

# 샘플 이미지 분석
sample_files = dicom_files[:100]  # 처음 100개 파일 분석
metadata_list = []

print("\n📊 샘플 이미지 메타데이터 수집 중...")
for f in tqdm(sample_files):
    try:
        _, meta = load_dicom_info(f)
        metadata_list.append(meta)
    except Exception as e:
        continue

meta_df = pd.DataFrame(metadata_list)
print("\n📋 샘플 메타데이터 요약:")

# 메타데이터 전처리 클래스
class MetadataPreprocessor:
    """메타데이터 전처리를 위한 클래스"""

    def __init__(self):
        self.sex_encoder = LabelEncoder()
        self.view_encoder = LabelEncoder()
        self.age_scaler = StandardScaler()
        self.size_scaler = StandardScaler()
        self.fitted = False

    def fit(self, df):
        """전처리기 학습"""
        # 범주형 변수 인코딩
        self.sex_encoder.fit(df['PatientSex'].fillna('U'))
        self.view_encoder.fit(df['ViewPosition'].fillna('PA'))

        # 수치형 변수 스케일링
        self.age_scaler.fit(df[['PatientAge']].fillna(50))
        self.size_scaler.fit(df[['Rows', 'Columns']].fillna(512))

        self.fitted = True
        return self

    def transform(self, df):
        """전처리 적용"""
        if not self.fitted:
            raise ValueError("Preprocessor must be fitted first!")

        result = pd.DataFrame()

        # 범주형 변수 인코딩
        result['sex_encoded'] = self.sex_encoder.transform(
            df['PatientSex'].fillna('U').apply(
                lambda x: x if x in self.sex_encoder.classes_ else 'U'
            )
        )
        result['view_encoded'] = self.view_encoder.transform(
            df['ViewPosition'].fillna('PA').apply(
                lambda x: x if x in self.view_encoder.classes_ else 'PA'
            )
        )

        # 수치형 변수 스케일링
        result['age_scaled'] = self.age_scaler.transform(
            df[['PatientAge']].fillna(50)
        )
        size_scaled = self.size_scaler.transform(
            df[['Rows', 'Columns']].fillna(512)
        )
        result['rows_scaled'] = size_scaled[:, 0]
        result['cols_scaled'] = size_scaled[:, 1]

        return result.values

    def fit_transform(self, df):
        """학습 후 변환"""
        return self.fit(df).transform(df)

# ResNet18 Feature Extractor 정의
class ResNetFeatureExtractor(nn.Module):
    """ResNet18 기반 이미지 특징 추출기"""

    def __init__(self, pretrained=True):
        super().__init__()

        # 사전학습된 ResNet18 로드
        resnet = models.resnet18(pretrained=pretrained)

        # 마지막 FC 레이어 제거
        self.features = nn.Sequential(*list(resnet.children())[:-1])

        # 출력 차원
        self.output_dim = 512

    def forward(self, x):
        x = self.features(x)
        x = x.view(x.size(0), -1)  # Flatten
        return x

def extract_features(model, dataloader):
    """모델을 사용하여 이미지 특징 추출"""
    model.eval()
    features = []
    indices = []

🔍 학습 포인트 #21

부트스트랩 방법을 사용하여 성능 지표의 95% 신뢰구간을 추정합니다. 이를 통해 성능 추정치의 불확실성을 정량화할 수 있습니다.

Code 
# 부트스트랩 신뢰구간 계산
def bootstrap_ci(y_true, y_prob, metric_func, n_bootstrap=1000, ci=0.95):
    """부트스트랩 신뢰구간 계산"""
    n_samples = len(y_true)
    scores = []

    for _ in range(n_bootstrap):
        indices = np.random.choice(n_samples, n_samples, replace=True)
        score = metric_func(y_true[indices], y_prob[indices])
        scores.append(score)

    lower = np.percentile(scores, (1 - ci) / 2 * 100)
    upper = np.percentile(scores, (1 + ci) / 2 * 100)

    return np.mean(scores), lower, upper

# 최종 앙상블 모델의 AUROC 신뢰구간
